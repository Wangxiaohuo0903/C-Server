# 推测式解码专用Drafter模型蒸馏方案

**目标**: 训练0.8B DeepSeek-Coder风格的Drafter，保持高接受率但推理更快

---

## 1. 蒸馏原理

### 为什么蒸馏有效？

传统训练:
```python
# 学生模型直接从数据学习
loss = CrossEntropy(student_output, ground_truth)
```

知识蒸馏:
```python
# 学生模型模仿教师的"软标签"
teacher_probs = softmax(teacher_logits / temperature)
student_probs = softmax(student_logits / temperature)

# 蒸馏损失：让学生的概率分布接近教师
distill_loss = KL_divergence(student_probs, teacher_probs)

# 总损失
loss = α * distill_loss + (1-α) * CrossEntropy(student_output, ground_truth)
```

**关键优势**:
- 教师的"软标签"包含更丰富的信息
- 不仅知道正确答案，还知道其他token的概率分布
- 学生可以学到教师的"思考方式"

### 推测式解码的特殊需求

普通蒸馏目标: 学生整体能力接近教师

推测式解码蒸馏目标: **学生的token分布尽可能接近教师**

```
关键差异:
- 不在乎学生单独生成质量
- 只在乎学生预测的下一个token是否和教师一致
- 接受率 > 生成质量
```

#### 3. 准备训练数据

#### 4. 蒸馏训练循环

```python
import torch
import torch.nn.functional as F
from torch.optim import AdamW
from torch.utils.data import DataLoader

# 超参数
temperature = 2.0      # 蒸馏温度 (2-4之间)
alpha = 0.7           # 蒸馏损失权重
learning_rate = 5e-5
batch_size = 8
gradient_accumulation_steps = 4
num_epochs = 3

optimizer = AdamW(student.parameters(), lr=learning_rate)
dataloader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)

student.train()

for epoch in range(num_epochs):
    for batch_idx, batch in enumerate(dataloader):
        input_ids = batch["input_ids"].to(device)
        attention_mask = batch["attention_mask"].to(device)

        # 教师前向传播 (无梯度)
        with torch.no_grad():
            teacher_outputs = teacher(
                input_ids=input_ids,
                attention_mask=attention_mask
            )
            teacher_logits = teacher_outputs.logits

        # 学生前向传播
        student_outputs = student(
            input_ids=input_ids,
            attention_mask=attention_mask
        )
        student_logits = student_outputs.logits

        # 1. 蒸馏损失 (KL散度)
        teacher_probs = F.softmax(teacher_logits / temperature, dim=-1)
        student_log_probs = F.log_softmax(student_logits / temperature, dim=-1)

        distill_loss = F.kl_div(
            student_log_probs,
            teacher_probs,
            reduction="batchmean"
        ) * (temperature ** 2)  # 温度平方缩放

        # 2. 标准交叉熵损失
        labels = input_ids[:, 1:]  # 下一个token预测
        student_logits_shifted = student_logits[:, :-1, :]
        ce_loss = F.cross_entropy(
            student_logits_shifted.reshape(-1, config.vocab_size),
            labels.reshape(-1),
            ignore_index=tokenizer.pad_token_id
        )

        # 3. 总损失
        loss = alpha * distill_loss + (1 - alpha) * ce_loss

        # 反向传播
        loss = loss / gradient_accumulation_steps
        loss.backward()

        if (batch_idx + 1) % gradient_accumulation_steps == 0:
            optimizer.step()
            optimizer.zero_grad()

        # 日志
        if batch_idx % 100 == 0:
            print(f"Epoch {epoch}, Batch {batch_idx}")
            print(f"  Distill loss: {distill_loss.item():.4f}")
            print(f"  CE loss: {ce_loss.item():.4f}")
            print(f"  Total loss: {loss.item():.4f}")

# 保存模型
student.save_pretrained("./deepseek-coder-0.8b-distilled")
tokenizer.save_pretrained("./deepseek-coder-0.8b-distilled")
```

#### 5. 推测式解码专门优化 (可选)

```python
# 额外优化：直接优化接受率
def speculative_aware_loss(student_logits, teacher_logits, input_ids):
    """
    推测式解码专用损失：惩罚top-1预测不匹配
    """
    # 教师的top-1预测
    teacher_top1 = teacher_logits.argmax(dim=-1)

    # 学生对教师top-1的概率
    student_probs = F.softmax(student_logits, dim=-1)
    teacher_top1_probs = torch.gather(
        student_probs,
        dim=-1,
        index=teacher_top1.unsqueeze(-1)
    ).squeeze(-1)

    # 负对数似然 (鼓励学生对教师选择赋高概率)
    spec_loss = -torch.log(teacher_top1_probs + 1e-10).mean()

    return spec_loss

# 在训练循环中添加
spec_loss = speculative_aware_loss(student_logits, teacher_logits, input_ids)
loss = alpha * distill_loss + (1-alpha) * ce_loss + 0.1 * spec_loss
```

   # 每1000步评估接受率
   @torch.no_grad()
   def evaluate_acceptance_rate(student, teacher, eval_data):
       total, accepted = 0, 0
       for batch in eval_data:
           teacher_pred = teacher(...).logits.argmax(-1)
           student_pred = student(...).logits.argmax(-1)
           accepted += (teacher_pred == student_pred).sum()
           total += teacher_pred.numel()
       return accepted / total
   ```

# 1. 基于TinyLlama添加LoRA

    r=16,  # LoRA秩
    lora_alpha=32,
    target_modules=["q_proj", "v_proj"],
    lora_dropout=0.05,
)

student = get_peft_model(
    AutoModelForCausalLM.from_pretrained("TinyLlama/TinyLlama-1.1B"),
    lora_config
)

# 2. 只训练LoRA参数 (14M vs 1.1B)
trainable_params = sum(p.numel() for p in student.parameters() if p.requires_grad)
print(f"可训练参数: {trainable_params / 1e6:.1f}M")  # ~14M

# 3. 蒸馏训练 (相同流程，但快10倍)

# 4. 合并LoRA
student = student.merge_and_unload()
```

**资源需求**: 1x A100 × 6小时 ≈ $20
